% \panel{name}{x position}{data key}{colour}{target radius}{title}{circle error}{extra drawing}
\newcommand{\panel}[8]{%
  \begin{axis}[name=#1, at={(#2,0)}, anchor=south west, width=8.6cm, height=8.6cm,
      axis equal image, xmin=-0.9, xmax=0.9, ymin=-0.9, ymax=0.9, axis lines=none, clip=false]
    \pgfplotsinvokeforeach{-0.8,-0.6,-0.4,-0.2,0,0.2,0.4,0.6,0.8}{
      \draw[saclaydk!13, line width=0.5pt] (axis cs:##1,-0.9) -- (axis cs:##1,0.9);
      \draw[saclaydk!13, line width=0.5pt] (axis cs:-0.9,##1) -- (axis cs:0.9,##1);
    }
    \addplot[draw=black!65, dashed, line width=1.4pt, domain=0:360, samples=120] ({#5*cos(x)}, {#5*sin(x)});
    \addplot[draw=cmove, line width=2.6pt] table[x=x, y=y] {data/top_#3_move.dat};
    \addplot[draw=cmove, line width=2.6pt] table[x=x, y=y] {data/top_#3_back.dat};
    \addplot[draw=#4, line width=2.6pt] table[x=x, y=y] {data/top_#3_circle.dat};
    % direction of travel
    \addplot[-{Stealth[length=4.5mm,width=3.5mm]}, draw=saclaydk, line width=1.8pt, domain=15:60, samples=30]
      ({(#5+0.16)*cos(x)}, {(#5+0.16)*sin(x)});
    \addplot[only marks, mark=*, mark size=4.5pt, saclaydk] coordinates {(0,0)};
    #8
  \end{axis}
  \node[anchor=south, align=center, text=#4] at ($(#1.north)+(0,0.05)$) {#6};
  \node[anchor=north, font=\normalsize, text=saclaydk] at ($(#1.south)+(0,0.05)$) {circle error #7 (mean)};
}
